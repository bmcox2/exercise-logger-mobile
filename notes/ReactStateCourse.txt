const [num, dispatch] = useReducer(numberReducer, initialState);

type numberAction = |
{ type: "INCREMENT" } |
{ type: "DECREMENT" } |
{ type: "RESET" } |
{ type: "SET"; value: number };

function numberReducer(num: number, action: numberAction): number {
    switch(action.type) {
        case "INCREMENT":
            return ++num;
        case "DECREMENT"
            return --num;
        case "RESET"
            num = 0;
            return num;
        case "SET"
            num = action.value;
            return num;
        default:
            return num;
    }
}

useReducer allows you to perform multiple different operations within a single function by just decribing what you want to do. useState only allows you to define one operation.

Quiz:
1. dispatch just calls the reducer and tells it what operation to do and passes the necassary parameters. It does not directly change the object, the reducer handles that.
2. If the user accidentally passes an action that does not exist, the reducer would have an error without the default. The default covers for this by just returning the same object without modifying it.
